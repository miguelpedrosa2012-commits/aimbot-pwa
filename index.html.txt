    <script type="module">
        import { AimEngine } from './engine.js';
        
        // Estado Global para o sistema
        window.state = {
            aimbot: false,
            headlock: false,
            filter: true,
            smoothing: 0.2,
            fov: 150
        };

        const engine = new AimEngine();

        window.toggle = (feature) => {
            window.state[feature] = !window.state[feature];
            document.getElementById(feature + '_toggle').classList.toggle('active');
        };

        document.getElementById('smoothing').oninput = (e) => {
            window.state.smoothing = e.target.value / 100;
        };

        document.getElementById('fov').oninput = (e) => {
            window.state.fov = e.target.value;
        };

        // --- LÓGICA DO CANVAS (FOV) ---
        const canvas = document.getElementById('overlay');
        const ctx = canvas.getContext('2d');
        let pulse = 0;

        function setupCanvas() {
            const dpr = window.devicePixelRatio || 1;
            canvas.width = window.innerWidth * dpr;
            canvas.height = window.innerHeight * dpr;
            ctx.scale(dpr, dpr);
            canvas.style.width = window.innerWidth + 'px';
            canvas.style.height = window.innerHeight + 'px';
        }

        function draw() {
            ctx.clearRect(0, 0, window.innerWidth, window.innerHeight);
            if (window.state.aimbot) {
                const centerX = window.innerWidth / 2;
                const centerY = window.innerHeight / 2;
                pulse += 0.04;
                const radius = parseFloat(window.state.fov) + Math.sin(pulse) * 3;

                ctx.beginPath();
                ctx.arc(centerX, centerY, radius, 0, Math.PI * 2);
                ctx.strokeStyle = 'rgba(255, 62, 62, 0.8)';
                ctx.lineWidth = 2;
                ctx.stroke();

                ctx.fillStyle = 'rgba(255, 62, 62, 0.9)';
                ctx.beginPath();
                ctx.arc(centerX, centerY, 2, 0, Math.PI * 2);
                ctx.fill();
            }
            requestAnimationFrame(draw);
        }

        setupCanvas();
        window.addEventListener('resize', setupCanvas);
        draw();

        if ('serviceWorker' in navigator) {
            navigator.serviceWorker.register('/sw.js');
        }
    </script>
